ADO_DEV

```sql
USE master;

-- Create database
CREATE DATABASE ado_dev;

-- Create login if it doesn't exist
IF NOT EXISTS (SELECT 1 FROM sys.server_principals WHERE name = 'ado_dev')
BEGIN
    CREATE LOGIN dms_dev WITH PASSWORD = 'Unit@032010', CHECK_EXPIRATION = OFF, CHECK_POLICY = OFF;
END;

-- Create user in DM01Y25DMS_DEV and grant permissions
USE ado_dev;
IF NOT EXISTS (SELECT 1 FROM sys.database_principals WHERE name = 'ado_dev,ies')
BEGIN
    CREATE USER ado_dev FOR LOGIN ado_dev;
	  CREATE USER ies FOR LOGIN ies;
END;

GRANT SELECT, INSERT, UPDATE, ALTER, DELETE, REFERENCES ON DATABASE::ado_dev TO ado_dev;
ALTER ROLE db_owner ADD MEMBER ies
```
