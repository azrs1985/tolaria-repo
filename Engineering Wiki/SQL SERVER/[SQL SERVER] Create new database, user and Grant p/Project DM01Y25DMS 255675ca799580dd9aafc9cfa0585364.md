DM01Y25DMS_DEV

```sql
USE master;

-- Create database
CREATE DATABASE DM01Y25DMS_DEV;

-- Create login if it doesn't exist
IF NOT EXISTS (SELECT 1 FROM sys.server_principals WHERE name = 'dms_dev')
BEGIN
    CREATE LOGIN dms_dev WITH PASSWORD = 'Unit@032010', CHECK_EXPIRATION = OFF, CHECK_POLICY = OFF;
END;

-- Create user in DM01Y25DMS_DEV and grant permissions
USE DM01Y25DMS_DEV;
IF NOT EXISTS (SELECT 1 FROM sys.database_principals WHERE name = 'dms_dev,ies')
BEGIN
    CREATE USER dms_dev FOR LOGIN dms_dev;
	  CREATE USER ies FOR LOGIN ies;
END;

GRANT SELECT, INSERT, UPDATE, ALTER, DELETE, REFERENCES ON DATABASE::DM01Y25DMS_DEV TO dms_dev;
ALTER ROLE db_owner ADD MEMBER ies;

-- Verify permissions
SELECT 
    p.name AS UserName,
    p.type_desc AS UserType,
    perm.permission_name AS Permission,
    perm.state_desc AS PermissionState,
    perm.class_desc AS PermissionClass,
    CASE 
        WHEN perm.class_desc = 'OBJECT' THEN OBJECT_NAME(perm.major_id)
        WHEN perm.class_desc = 'SCHEMA' THEN SCHEMA_NAME(perm.major_id)
        WHEN perm.class_desc = 'DATABASE' THEN DB_NAME()
        ELSE CAST(perm.major_id AS nvarchar(128))
    END AS Scope
FROM sys.database_principals p
LEFT JOIN sys.database_permissions perm 
    ON p.principal_id = perm.grantee_principal_id
WHERE p.name = 'dms_dev'
ORDER BY perm.permission_name;

-- Verify roles
SELECT DP.name AS UserName, DR.name AS RoleName
FROM sys.database_role_members DRM
JOIN sys.database_principals DR ON DRM.role_principal_id = DR.principal_id
JOIN sys.database_principals DP ON DRM.member_principal_id = DP.principal_id
WHERE DP.name = 'ies';
```
