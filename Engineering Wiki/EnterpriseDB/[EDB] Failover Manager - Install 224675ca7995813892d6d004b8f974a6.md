### INSTALL ON BOTH MASTER AND SLAVE

```bash
[root@]# curl -1sLf 'https://downloads.enterprisedb.com/vpxyZhvk35DAognci2N7cld4g0UfcDAK/enterprise/setup.rpm.sh' | sudo -E bash
[root@]# dnf -y install edb-as16-server edb-efm410 java-17-openjdk
[root@]# firewall-cmd --permanent --add-port={5444,7800}/tcp
[root@]# firewall-cmd --permanent --remove-service={cockpit,dhcpv6-client}
[root@]# firewall-cmd --reload
[root@]# mkdir -p /usr/edb/as16/data
[root@]# chown enterprisedb:enterprisedb -R /usr/edb/as16/data
[root@]# chmod 0700 /usr/edb/as16/data
[root@]# rm -rf /var/lib/edb/as16/data
[root@]# sudo -u enterprisedb ln -s /usr/edb/as16/data/ /var/lib/edb/as16/data
[root@]# sudo -u enterprisedb /usr/edb/as16/bin/initdb -D /usr/edb/as16/data/
[root@]# systemctl daemon-reload
[root@]# systemctl start edb-as-16.service
[root@]# systemctl status edb-as-16.service
[root@]# psql -d edb -U enterprisedb
edb=# CREATE USER repl WITH REPLICATION ENCRYPTED PASSWORD 'repl';
edb=# \password enterprisedb
Enter new password for user "enterprisedb":
Enter it again:
edb=# \q
```

### ON SLAVE NODE

```bash
[root@]# rm -rf /usr/edb/as16/data/
[root@]# pg_basebackup -h 192.168.11.26 -U repl -p 5444 -D /usr/edb/as16/data -Fp -Xs -P -R
[root@]# chown enterprisedb:enterprisedb -R /usr/edb/as16/data
[root@]# chmod 0700 /usr/edb/as16/data
[root@]# systemctl daemon-reload
[root@]# systemctl start edb-as-16
[root@]# cat /usr/edb/as16/data/postgresql.auto.conf
```

## Do not edit this file manually!

It will be overwritten by the ALTER SYSTEM command.

```bash
primary_conninfo = 'user=repl passfile=''/root/.pgpass''
channel_binding=prefer host=192.168.11.26 port=5444 sslmode=prefer
sslcompression=0 sslcertmode=allow sslsni=1
ssl_min_protocol_version=TLSv1.2 gssencmode=prefer krbsrvname=postgres
gssdelegation=0 target_session_attrs=any load_balance_hosts=disable'
```
